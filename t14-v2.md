The goals of the second version:

* Add a USB hub
* Add a 4 degree typing angle

For the USB hub, I am using an Adafruit CH334F breakout board. I am exposing two type A ports and one type C port.

For the MCU, I am using a teensy 2.0 dev board. It's based on ATmega32U4. It breaks out 25 pins. With the compressed matrix that uses 22 pins, the two pins for PS/2 and one for the caps lock indicator, that's just enough.

I am also taking advantage of the usart implementation of PS/2 in QMK. Note that the clock and data pins have to be `PD5` and `PD2`.

Here's an excerpt of the keyboard definition.

```
{
  "matrix_pins": {
    "rows": ["F0", "F1", "F4", "F5", "F6", "F7", "B6", "B5", "B4", "D7", "D6", "C7", "C6", "D4"],
    "cols": ["B0", "B1", "B2", "B3", "B7", "D0", "D1", "D3"]
  },
  "indicators": {
    "caps_lock": "E6",
    "on_state": 0
  },
  "ps2": {
    "enabled": true,
    "mouse_enabled": true,
    "clock_pin": "D5",
    "data_pin": "D2",
    "driver": "usart"
  }
}
```

And in `config.h`, we need to define:

```
#pragma once

// clang-format off
#ifdef PS2_DRIVER_INTERRUPT
#define PS2_INT_INIT()  do {    \
    EICRA |= ((1<<ISC21) |      \
              (0<<ISC20));      \
} while (0)
#define PS2_INT_ON()  do {      \
    EIMSK |= (1<<INT2);         \
} while (0)
#define PS2_INT_OFF() do {      \
    EIMSK &= ~(1<<INT2);        \
} while (0)
#define PS2_INT_VECT   INT2_vect
#endif

#ifdef PS2_DRIVER_USART
/* synchronous, odd parity, 1-bit stop, 8-bit data, sample at falling edge */
/* set DDR of CLOCK as input to be slave */
#define PS2_USART_INIT() do {   \
    PS2_CLOCK_DDR &= ~(1<<PS2_CLOCK_BIT);   \
    PS2_DATA_DDR &= ~(1<<PS2_DATA_BIT);     \
    UCSR1C = ((1 << UMSEL10) |  \
              (3 << UPM10)   |  \
              (0 << USBS1)   |  \
              (3 << UCSZ10)  |  \
              (0 << UCPOL1));   \
    UCSR1A = 0;                 \
    UBRR1H = 0;                 \
    UBRR1L = 0;                 \
} while (0)
#define PS2_USART_RX_INT_ON() do {  \
    UCSR1B = ((1 << RXCIE1) |       \
              (1 << RXEN1));        \
} while (0)
#define PS2_USART_RX_POLL_ON() do { \
    UCSR1B = (1 << RXEN1);          \
} while (0)
#define PS2_USART_OFF() do {    \
    UCSR1C = 0;                 \
    UCSR1B &= ~((1 << RXEN1) |  \
                (1 << TXEN1));  \
} while (0)
#define PS2_USART_RX_READY      (UCSR1A & (1<<RXC1))
#define PS2_USART_RX_DATA       UDR1
#define PS2_USART_ERROR         (UCSR1A & ((1<<FE1) | (1<<DOR1) | (1<<UPE1)))
#define PS2_USART_RX_VECT       USART1_RX_vect
#endif

// clang-format on
```

Note: use `sudo` to flash qmk. The bootloader `halfkay` is quirky this way.
